Q2 Explain the bias-variance tradeoff. ▼
ANSWER: - Bias: Error from wrong assumptions (underfitting) - Variance: Error from sensitivity to training data (overfitting) - Goal: Balance both for optimal generalization - High Bias: Model too simple - High Variance: Model too complex
from sklearn.tree import DecisionTreeRegressor
from sklearn.datasets import make_regression

def demonstrate_bias_variance():
    X, y = make_regression(n_samples=100, n_features=1, noise=10, random_state=42)
    
    # High bias (underfitting)
    model_bias = DecisionTreeRegressor(max_depth=1)
    model_bias.fit(X, y)
    
    # High variance (overfitting)
    model_variance = DecisionTreeRegressor(max_depth=20)
    model_variance.fit(X, y)
    
    # Balanced
    model_balanced = DecisionTreeRegressor(max_depth=5)
    model_balanced.fit(X, y)
    
    print("Q2 Bias-Variance: Models trained with different complexities")

demonstrate_bias_variance()

Q7 Explain gradient descent and its variants. ▼
ANSWER: - Optimization algorithm to minimize loss function - Updates weights: w = w - learning_rate * gradient - Variants: * Batch GD: Uses all data * Stochastic GD (SGD): Uses one sample * Mini-batch GD: Uses batch of samples
def gradient_descent_demo():
    # Minimize f(x) = x^2
    x = 10.0  # Starting point
    learning_rate = 0.1
    iterations = 20
    
    history = [x]
    for i in range(iterations):
        gradient = 2 * x  # Derivative of x^2
        x = x - learning_rate * gradient
        history.append(x)
    
    print(f"Q7 Gradient Descent: Start=10, End={x:.6f}, Iterations={iterations}")
    return history

gradient_descent_demo()
Q8 What is batch normalization? ▼

# Conceptual implementation
def batch_norm_concept(X, epsilon=1e-5):
    mean = np.mean(X, axis=0)
    var = np.var(X, axis=0)
    X_norm = (X - mean) / np.sqrt(var + epsilon)
    return X_norm

Q9 What is dropout? ▼
ANSWER: - Regularization technique for neural networks - Randomly drops neurons during training - Prevents co-adaptation of neurons - Typical rate: 0.2-0.5 - Only applied during training, not inference
def dropout_demo(X, dropout_rate=0.5):
    # During training
    mask = np.random.binomial(1, 1-dropout_rate, size=X.shape)
    X_dropped = X * mask / (1 - dropout_rate)  # Scale to maintain expected value
    return X_dropped

X = np.ones((5, 10))
X_after_dropout = dropout_demo(X, dropout_rate=0.5)

Q10 Compare different activation functions. ▼
ANSWER: - Sigmoid: σ(x) = 1/(1+e^-x), outputs [0,1], suffers from vanishing gradients - Tanh: tanh(x), outputs [-1,1], zero-centered - ReLU: max(0,x), solves vanishing gradient, can have dying neurons - Leaky ReLU: max(αx, x), fixes dying ReLU - Softmax: Used for multi-class classification output
def activation_functions(x):
    sigmoid = 1 / (1 + np.exp(-x))
    tanh = np.tanh(x)
    relu = np.maximum(0, x)
    leaky_relu = np.where(x > 0, x, 0.01 * x)
    
    return {
        'sigmoid': sigmoid,
        'tanh': tanh,
        'relu': relu,
        'leaky_relu': leaky_relu
    }

x = np.array([-2, -1, 0, 1, 2])
activations = activation_functions(x)

print(f"Q10 Activation for x={x}:")
for name, values in activations.items():
    print(f" {name}: {values}")
Q11 What are common loss functions? ▼
ANSWER: - MSE (Mean Squared Error): Regression, penalizes large errors - MAE (Mean Absolute Error): Regression, robust to outliers - Cross-Entropy: Classification, measures probability distribution difference - Hinge Loss: SVM, margin-based
def loss_functions_demo():
    y_true = np.array([3.0, -0.5, 2.0, 7.0])
    y_pred = np.array([2.5, 0.0, 2.0, 8.0])
    
    mse = np.mean((y_true - y_pred) ** 2)
    mae = np.mean(np.abs(y_true - y_pred))
    
    print(f"Q11 Loss Functions:")
    print(f" MSE: {mse:.4f}")
    print(f" MAE: {mae:.4f}")
    
    # Binary cross-entropy
    y_true_binary = np.array([1, 0, 1, 1])
    y_pred_proba = np.array([0.9, 0.1, 0.8, 0.7])
    
    bce = -np.mean(y_true_binary * np.log(y_pred_proba + 1e-7) +
                   (1 - y_true_binary) * np.log(1 - y_pred_proba + 1e-7))
    print(f" Binary Cross-Entropy: {bce:.4f}")

# Conceptual CNN operation
def conv2d_simple(image, kernel):
    """Simple 2D convolution"""
    h, w = image.shape
    kh, kw = kernel.shape
    output = np.zeros((h - kh + 1, w - kw + 1))
    
    for i in range(output.shape[0]):
        for j in range(output.shape[1]):
            output[i, j] = np.sum(image[i:i+kh, j:j+kw] * kernel)
    
    return output

Q34 What is attention mechanism? ▼
ANSWER: - Allows model to focus on relevant parts of input - Computes weighted sum of input values - Attention score = similarity(query, key) - Used in Transformers - Types: Self-attention, Multi-head attention - Formula: Attention(Q,K,V) = softmax(QK^T/√d_k)V
def simple_attention(query, keys, values):
    """Simple attention mechanism"""
    # Compute attention scores
    scores = np.dot(keys, query)
    
    # Softmax
    attention_weights = np.exp(scores) / np.sum(np.exp(scores))
    
    # Weighted sum of values
    output = np.dot(attention_weights, values)
    
    return output, attention_weights

Q40 What is early stopping? ▼
ANSWER: - Regularization technique - Stops training when validation loss stops improving - Prevents overfitting - Parameters: * Patience: Epochs to wait * Min_delta: Minimum improvement threshold - Saves best model checkpoint
def early_stopping_demo():
    """Simulate early stopping"""
    val_losses = [0.8, 0.6, 0.5, 0.45, 0.44, 0.44, 0.45, 0.46]
    patience = 3
    best_loss = float('inf')
    patience_counter = 0
    
    for epoch, loss in enumerate(val_losses):
        if loss < best_loss:
            best_loss = loss
            patience_counter = 0
        else:
            patience_counter += 1
        
        if patience_counter >= patience:
            print(f"Q40 Early Stopping:")
            print(f" Stopped at epoch {epoch + 1}")
            print(f" Best validation loss: {best_loss:.3f}")
            break

early_stopping_demo()
Q41 Difference between normalization and standardization? ▼
ANSWER: Normalization (Min-Max): - Scales to [0, 1]: (x - min) / (max - min) - Preserves relationships - Sensitive to outliers Standardization (Z-score): - Centers around 0: (x - μ) / σ - Not bounded - Robust to outliers - Better for Gaussian distributions
X_sample = np.array([[1, 2], [3, 4], [5, 6], [100, 8]])

# Normalization
X_norm = (X_sample - X_sample.min(axis=0)) / (X_sample.max(axis=0) - X_sample.min(axis=0))

# Standardization
X_std = (X_sample - X_sample.mean(axis=0)) / X_sample.std(axis=0)

print(f"Q41 Normalization vs Standardization:")
print(f" Normalized range: [{X_norm.min():.2f}, {X_norm.max():.2f}]")
print(f" Standardized mean: {X_std.mean():.2f}, std: {X_std.std():.2f}")

from scipy import stats

# Simulate A/B test results
conversions_A = np.random.binomial(1, 0.10, 1000)  # 10% conversion
conversions_B = np.random.binomial(1, 0.12, 1000)  # 12% conversion

# Chi-square test
contingency_table = np.array([
    [conversions_A.sum(), len(conversions_A) - conversions_A.sum()],
    [conversions_B.sum(), len(conversions_B) - conversions_B.sum()]
])

chi2, p_value, dof, expected = stats.chi2_contingency(contingency_table)
